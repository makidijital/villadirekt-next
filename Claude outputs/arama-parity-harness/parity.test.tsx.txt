function maybeErr(key: string) {
  const m = S.err[key];
  if (m === 'throw') throw new Error('sim-throw:' + key);
  if (m === 'error') return { data: null, error: { message: 'sim-error:' + key } };
  return null;
}
async function db<T>(name: string, fn: () => T): Promise<T> {
  S.calls.push(name);
  S.inFlight++;
  S.maxInFlight = Math.max(S.maxInFlight, S.inFlight);
  try {
    if (S.lat > 0) await new Promise((r) => setTimeout(r, S.lat));
    const out = fn();
    try { S.bytes[name] = (S.bytes[name] || 0) + JSON.stringify((out as any)?.data ?? out).length; } catch {}
    return out;
  } finally {
    S.inFlight--;
  }
}

async function renderOnce(sp: SP, locale: "tr" | "en" | "de") {
  const { default: AramaPageBody } = await import("@/app/components/search/AramaPageBody");
  const el = await AramaPageBody({ locale, searchParams: Promise.resolve(sp) as any });
  const { container } = render(el);
  const html = container.innerHTML;
  cleanup();
  return html;
}

function summarize(html: string) {
  const cards = [...html.matchAll(/data-card="([^"]*)"/g)].map((m) =>
    JSON.parse(m[1].replace(/&quot;/g, '"').replace(/&amp;/g, "&").replace(/&lt;/g, "<").replace(/&gt;/g, ">"))
  );
  const side = html.match(/data-sidebar="([^"]*)"/);
  const sidebar = side ? JSON.parse(side[1].replace(/&quot;/g, '"').replace(/&amp;/g, "&")) : null;
  return {
    total: sidebar?.resultCount ?? null,
    cards: cards.filter((c) => !c.isFlexible).map((c) => c.id.slice(-4)),
    flexible: cards.filter((c) => c.isFlexible).map((c) => c.id.slice(-4)),
  };
}

const MODE = process.env.PARITY_MODE || "compare";
const OUT = process.env.PARITY_OUT || path.join(__dirname, "out");
fs.mkdirSync(OUT, { recursive: true });

describe(`/arama parity (${MODE})`, () => {
  if (MODE === "perf") {
    it("perf", async () => {
      const N = Number(process.env.PARITY_N || 1500);
      S.fx = buildFixture(N);
      const lat = Number(process.env.PARITY_LAT || 0);
      const keys = ["01-bos", "02-tarih", "07-hepsi", "08-esnek3", "09-fiyat-artan", "13-tarihsiz-fiyat-artan", "14-kapasite-artan", "16-EN-hepsi", "16b-EN-fiyat-esnek"];
      const res: any = { N, lat, scenarios: {} };
      for (const k of keys) {
        const sc = SCENARIOS.find((s) => s.key === k)!;
        S.currency = sc.currency;
        S.err = {};
        S.lat = lat;
        const times: number[] = [];
        let calls: string[] = [];
        let maxIn = 0;
        let bytes = {};
        for (let r = 0; r < 5; r++) {
          S.calls = []; S.bytes = {}; S.maxInFlight = 0;
          const t0 = performance.now();
          await renderOnce(sc.sp, sc.locale ?? "tr");
          times.push(performance.now() - t0);
          calls = S.calls.slice(); maxIn = S.maxInFlight; bytes = { ...S.bytes };
        }
        times.sort((a, b) => a - b);
        res.scenarios[k] = {
          medianMs: Math.round(times[2] * 10) / 10,
          minMs: Math.round(times[0] * 10) / 10,
          dbCalls: calls.filter((c) => !c.startsWith("cache:")).length,
          calls,
          maxConcurrentDb: maxIn,
          bytes,
        };
      }
      fs.writeFileSync(path.join(OUT, `perf-${process.env.PARITY_LABEL || "x"}-N${N}-L${lat}.json`), JSON.stringify(res, null, 1));
    });
    return;
  }

  it("scenarios", async () => {
    const N = Number(process.env.PARITY_N || 90);
    S.fx = buildFixture(N);
    S.lat = 0;
    const result: Record<string, { html: string; calls: string[]; summary: any }> = {};
    for (const sc of SCENARIOS) {
      S.currency = sc.currency;
      S.err = sc.err ?? {};
      const pages = sc.pages ?? 8;
      const pageList = pages === 0 ? [undefined] : Array.from({ length: pages }, (_, i) => String(i + 1));
      for (const p of pageList) {
        S.calls = [];
        const sp = p ? { ...sc.sp, page: p } : sc.sp;
        const html = await renderOnce(sp, sc.locale ?? "tr");
        result[`${sc.key}|p${p ?? "-"}`] = { html, calls: S.calls.slice(), summary: summarize(html) };
      }
    }
    const file = path.join(OUT, `baseline-N${N}.json`);
    if (MODE === "baseline") {
      fs.writeFileSync(file, JSON.stringify(result, null, 1));
      return;
    }
    const base = JSON.parse(fs.readFileSync(file, "utf8"));
    const diffs: string[] = [];
    const callDiffs: Record<string, { before: string[]; after: string[] }> = {};
    for (const k of Object.keys(base)) {
      if (!result[k]) { diffs.push(`${k}: yok`); continue; }
      if (base[k].html !== result[k].html) diffs.push(k);
      if (JSON.stringify(base[k].calls) !== JSON.stringify(result[k].calls)) callDiffs[k] = { before: base[k].calls, after: result[k].calls };
    }
    fs.writeFileSync(
      path.join(OUT, `compare-${process.env.PARITY_LABEL || "x"}-N${N}.json`),
      JSON.stringify({ htmlDiffs: diffs, compared: Object.keys(base).length, callDiffs }, null, 1)
    );
    expect(diffs).toEqual([]);
  });
});
